# SQLite DB 조회
## 한 줄 조회 (큰따옴표 안에 작은따옴표)

### 행 수
python -c "import sqlite3; c=sqlite3.connect('data/scan.db'); print(c.execute('SELECT COUNT(*) FROM matches').fetchone()[0])"

### 패턴별 집계
python -c "import sqlite3; c=sqlite3.connect('data/scan.db'); [print(f'{n:16} {k}') for n,k in c.execute('SELECT pattern_name, COUNT(*) FROM matches GROUP BY pattern_name ORDER BY 2 DESC')]"

### collect 위치별 집계
python -c "import sqlite3; c=sqlite3.connect('data/scan.db'); [print(f'{l:14} {k}') for l,k in c.execute('SELECT location, COUNT(*) FROM collect GROUP BY location ORDER BY 2 DESC')]"

### 스캔 이력
python -c "import sqlite3; c=sqlite3.connect('data/scan.db'); [print(r) for r in c.execute('SELECT id, status, chunks_scanned, finished_at FROM scans ORDER BY id DESC LIMIT 5')]"
